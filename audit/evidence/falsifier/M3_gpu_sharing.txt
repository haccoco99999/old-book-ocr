# testeval: end-to-end starts the version on EVERY worker port
def serve_on(name: str, ports: list[int]) -> None:
    """One model at a time on every worker port, after the artefact re-check of serve.sh."""
    version, _, row = serving.preflight(name)
    for s in servers.recognizers():
        if s["serves"] != name or s["port"] not in ports:
            servers.stop(s["port"])
            log(f"stopped :{s['port']} ({s['serves']})")
    running = {s["port"] for s in servers.recognizers()}
    spec = "base" if version["model_spec"] == "base" else str(paths.resolve(version["run_dir"]))
    for port in ports:
        if port not in running:
            servers.start_recognizer(spec, name, port)
            log(f"started {name} on :{port}")
