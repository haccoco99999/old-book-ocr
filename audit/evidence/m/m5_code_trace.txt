    64	def append_log(path: Path, entry: dict) -> dict:
    65	    """Append-only. Each line carries the sha256 of the line before it (a hash chain; git history is the
    66	    tamper evidence that matters)."""
    67	    lines = path.read_text(encoding="utf-8").splitlines() if path.exists() else []
    68	    prev = hashlib.sha256(lines[-1].encode()).hexdigest() if lines else None
    69	    entry = {"at": now(), **entry, "prev_line_sha256": prev}
    70	    with open(path, "a", encoding="utf-8") as fh:
    71	        fh.write(json.dumps(entry, ensure_ascii=False, sort_keys=False) + "\n")
    72	        fh.flush()
    73	        os.fsync(fh.fileno())
    74	    return entry
